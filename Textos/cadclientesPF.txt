// ── Aplica máscara enquanto o usuário digita ──────────────

function mascararCPF(e) {
    let v = e.target.value.replace(/\D/g, '').slice(0, 11);
    if (v.length > 9)      v = v.replace(/(\d{3})(\d{3})(\d{3})(\d{1,2})/, '$1.$2.$3-$4');
    else if (v.length > 6) v = v.replace(/(\d{3})(\d{3})(\d{1,3})/, '$1.$2.$3');
    else if (v.length > 3) v = v.replace(/(\d{3})(\d{1,3})/, '$1.$2');
    e.target.value = v;
}

function mascararRG(e) {
    // Formato RS: 00.000.000-0 (10 dígitos)
    let v = e.target.value.replace(/\D/g, '').slice(0, 10);
    if (v.length > 8)      v = v.replace(/(\d{2})(\d{3})(\d{3})(\d{1,2})/, '$1.$2.$3-$4');
    else if (v.length > 5) v = v.replace(/(\d{2})(\d{3})(\d{1,3})/, '$1.$2.$3');
    else if (v.length > 2) v = v.replace(/(\d{2})(\d{1,3})/, '$1.$2');
    e.target.value = v;
}

function mascararTelefone(e) {
    // (99) 3333-3333  — 8 dígitos no número
    let v = e.target.value.replace(/\D/g, '').slice(0, 10);
    if (v.length > 6)      v = v.replace(/(\d{2})(\d{4})(\d{1,4})/, '($1) $2-$3');
    else if (v.length > 2) v = v.replace(/(\d{2})(\d{1,4})/, '($1) $2');
    else if (v.length > 0) v = v.replace(/(\d{1,2})/, '($1');
    e.target.value = v;
}

function mascararCelular(e) {
    // (99) 99999-9999 — 9 dígitos no número
    let v = e.target.value.replace(/\D/g, '').slice(0, 11);
    if (v.length > 7)      v = v.replace(/(\d{2})(\d{5})(\d{1,4})/, '($1) $2-$3');
    else if (v.length > 2) v = v.replace(/(\d{2})(\d{1,5})/, '($1) $2');
    else if (v.length > 0) v = v.replace(/(\d{1,2})/, '($1');
    e.target.value = v;
}

function mascararCEP(e) {
    // 00000-000
    let v = e.target.value.replace(/\D/g, '').slice(0, 8);
    if (v.length > 5) v = v.replace(/(\d{5})(\d{1,3})/, '$1-$2');
    e.target.value = v;
}

// ── Validação de CPF no blur ──────────────────────────────

document.addEventListener('DOMContentLoaded', () => {
    const aplicar = (id, fn) => {
        const el = document.getElementById(id);
        if (el) el.addEventListener('input', fn);
    };

    aplicar('cpf',      mascararCPF);
    aplicar('rg',       mascararRG);
    aplicar('telefone', mascararTelefone);
    aplicar('celular',  mascararCelular);
    aplicar('cep',      mascararCEP);

    // Validação CPF no blur
    const elCpf = document.getElementById('cpf');
    if (elCpf) {
        elCpf.addEventListener('blur', function () {
            const valor = this.value.trim();
            if (valor === '') return;
            if (!validarCPF(valor)) {
                alert('CPF inválido! Por favor, corrija os dados.');
                setTimeout(() => { this.focus(); this.select(); }, 10);
            }
        });
    }
});

<script>

    function calcularDigito(base, pesoInicial) {
        let soma = 0;
        let peso = pesoInicial;
        for (let i = 0; i < base.length; i++) {
            soma += parseInt(base.charAt(i)) * peso;
            peso = (peso < 2) ? 9 : peso - 1;
            }
        const resto = soma % 11;
        return resto < 2 ? 0 : 11 - resto;
    }

const STORAGE_KEY = 'PDV_CLIENTES_PF';
const EDIT_KEY = 'CLIENTE_EDITAR';

////////////////////////////////////////////////////////////////
// CPF VALIDATOR
////////////////////////////////////////////////////////////////
function validarCPF(cpf) {
    const numeros = cpf.replace(/\D/g, '');

    if (numeros.length !== 11) return false;
    if (/^(\d)\1+$/.test(numeros)) return false;

    let soma = 0;

    for (let i = 0; i < 9; i++) {
        soma += parseInt(numeros.charAt(i)) * (10 - i);
    }

    let resto = soma % 11;
    let d1 = resto < 2 ? 0 : 11 - resto;

    soma = 0;

    for (let i = 0; i < 10; i++) {
        soma += parseInt(numeros.charAt(i)) * (11 - i);
    }

    resto = soma % 11;
    let d2 = resto < 2 ? 0 : 11 - resto;

    return d1 === parseInt(numeros[9]) && d2 === parseInt(numeros[10]);
}

////////////////////////////////////////////////////////////////
// STATE

let formularioAlterado = false;

////////////////////////////////////////////////////////////////
// INIT EDIT MODE (ÚNICO E DEFINITIVO)
////////////////////////////////////////////////////////////////
function carregarEdicao() {

    const editData = localStorage.getItem(EDIT_KEY);
    if (!editData) return;

    const cliente = JSON.parse(editData);

    const form = document.getElementById('formClientePF');
    if (form) form.dataset.editId = cliente.id;

    const set = (id, val) => {
        const el = document.getElementById(id);
        if (el) el.value = val ?? '';
    };

    set('nomeCompleto', cliente.nomeCompleto);
    set('cpf', cliente.cpf);
    set('dataNascimento', cliente.dataNascimento);
    set('rg', cliente.rg);
    set('sexo', cliente.sexo);
    set('telefone', cliente.telefone);
    set('celular', cliente.celular);
    set('email', cliente.email);
    set('cep', cliente.cep);
    set('rua', cliente.rua);
    set('numero', cliente.numero);
    set('bairro', cliente.bairro);
    set('cidade', cliente.cidade);
    set('estado', cliente.estado);
    set('observacoes', cliente.observacoes);

    localStorage.removeItem(EDIT_KEY);
}

// LIMPAR FORM

function limparFormulario() {

    const ids = [
        'nomeCompleto','cpf','dataNascimento','rg','sexo',
        'telefone','celular','email','cep','rua',
        'numero','bairro','cidade','estado','observacoes'
    ];

    ids.forEach(id => {
        const el = document.getElementById(id);
        if (el) el.value = '';
    });

    const form = document.getElementById('formClientePF');
    if (form) delete form.dataset.editId;

    const first = document.getElementById('nomeCompleto');

    window.scrollTo({
        top: 0,
        behavior: 'smooth'

    setTimeout(() => {

    }, 300);
}

////////////////////////////////////////////////////////////////
// GERAR ID
////////////////////////////////////////////////////////////////
function gerarId() {
    const lista = JSON.parse(localStorage.getItem(STORAGE_KEY) || '[]');
    return lista.length ? Math.max(...lista.map(c => Number(c.id)||0)) + 1 : 1;
}

////////////////////////////////////////////////////////////////
// SALVAR
////////////////////////////////////////////////////////////////
function salvarCadastro(e) {
    e.preventDefault();

    const cpf = document.getElementById('cpf').value.trim();

    if (!cpf || !validarCPF(cpf)) {
        alert('CPF inválido');
        return;
    }

    const form = document.getElementById('formClientePF');
    const editId = form?.dataset.editId ? Number(form.dataset.editId) : null;

    const lista = JSON.parse(localStorage.getItem(STORAGE_KEY) || '[]');
    const agora = new Date().toISOString();

    const cliente = {
        id: editId || gerarId(),
        nomeCompleto: document.getElementById('nomeCompleto').value,
        cpf,
        dataNascimento: document.getElementById('dataNascimento').value,
        rg: document.getElementById('rg').value,
        sexo: document.getElementById('sexo').value,
        telefone: document.getElementById('telefone').value,
        celular: document.getElementById('celular').value,
        email: document.getElementById('email').value,
        cep: document.getElementById('cep').value,
        rua: document.getElementById('rua').value,
        numero: document.getElementById('numero').value,
        bairro: document.getElementById('bairro').value,
        cidade: document.getElementById('cidade').value,
        estado: document.getElementById('estado').value,
        observacoes: document.getElementById('observacoes').value,
        dataAlteracao: agora
    };

    if (editId) {
        const idx = lista.findIndex(c => c.id === editId);
        if (idx !== -1) {
            cliente.dataCadastro = lista[idx].dataCadastro;
            lista[idx] = cliente;
        }
    } else {
        cliente.dataCadastro = agora;
        lista.push(cliente);
    }

    localStorage.setItem(STORAGE_KEY, JSON.stringify(lista));

    formularioAlterado = false;

    const msg = editId ? 'Cliente atualizado com sucesso!' : 'Cliente salvo com sucesso!';
    if (confirm(msg + '\n\nDeseja voltar à tabela de clientes?')) {
        window.location.href = 'tabClientePF.html';
    } else {
        limparFormulario();
    }
}

function baixarJSON() {

    let clientes = JSON.parse(localStorage.getItem(STORAGE_KEY) || '[]');

    const json = JSON.stringify(clientes, null, 2);

    const blob = new Blob([json], { type: 'application/json' });

    const url = URL.createObjectURL(blob);

    const a = document.createElement('a');
    a.href = url;
    a.download = 'cadclientesPF.json';

    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);

    URL.revokeObjectURL(url);
}

function importarPrimeiroCliente(e) {
    const file = e.target.files[0];
    if (!file) return;

    const reader = new FileReader();
    reader.onload = function () {
        try {
            let dados = JSON.parse(reader.result);

            // aceita objeto único ou array
            if (!Array.isArray(dados)) dados = [dados];

            if (!dados.length) {
                alert('O arquivo JSON está vazio.');
                return;
            }

            if (formularioAlterado) {
                if (!confirm('Descartar alterações atuais e carregar o JSON?')) return;
            }

            const c = dados[0];

            const form = document.getElementById('formClientePF');
            if (form) form.dataset.editId = c.id || '';

            const set = (id, val) => {
                const el = document.getElementById(id);
                if (el) el.value = val ?? '';
            };

            set('nomeCompleto',   c.nomeCompleto);
            set('cpf',            c.cpf);
            set('dataNascimento', c.dataNascimento);
            set('rg',             c.rg);
            set('sexo',           c.sexo);
            set('telefone',       c.telefone);
            set('celular',        c.celular);
            set('email',          c.email);
            set('cep',            c.cep);
            set('rua',            c.rua);
            set('numero',         c.numero);
            set('bairro',         c.bairro);
            set('cidade',         c.cidade);
            set('estado',         c.estado);
            set('observacoes',    c.observacoes);

            formularioAlterado = false;

            window.scrollTo({ top: 0, behavior: 'smooth' });
            setTimeout(() => document.getElementById('nomeCompleto')?.focus(), 300);

            alert(`Cliente "${c.nomeCompleto || 'sem nome'}" carregado com sucesso!`);
        } catch (err) {
            alert('Erro ao ler o JSON: ' + err.message);
        }
        // limpa o input para permitir reimportar o mesmo arquivo
        e.target.value = '';
    };
    reader.readAsText(file);
}
