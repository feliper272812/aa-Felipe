// =========================================================
//  tabFornecedores.js — INLINE (evita problemas de caminho)
//  Chave localStorage: PDV_FORNECEDORES  (mesma do cadFornecedores.html)
//  Edição via:         FORNECEDOR_EDITAR (mesma do cadFornecedores.html)
// =========================================================

const STORAGE_KEY = "PDV_FORNECEDORES";
const EDIT_KEY    = "FORNECEDOR_EDITAR";

// ── Diagnóstico de arranque ───────────────────────────────
// Abre o console (F12) para ver o que está no localStorage
console.log("[tabFornecedores] chave:", STORAGE_KEY);
console.log("[tabFornecedores] raw  :", localStorage.getItem(STORAGE_KEY));

// ── Utilitários ───────────────────────────────────────────

function getFornecedores() {
  try {
    return JSON.parse(localStorage.getItem(STORAGE_KEY)) || [];
  } catch {
    return [];
  }
}

function saveFornecedores(lista) {
  localStorage.setItem(STORAGE_KEY, JSON.stringify(lista));
}

function formatarData(iso) {
  if (!iso) return "—";
  const d = new Date(iso.includes("T") ? iso : iso + "T00:00:00");
  if (isNaN(d)) return iso;
  return d.toLocaleDateString("pt-BR");
}

function escaparHTML(str) {
  if (str == null) return "";
  return String(str)
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;");
}

// ── Renderização ──────────────────────────────────────────

function renderizarTabela(lista) {
  const tbody   = document.getElementById("tbodyFornecedores");
  const totalEl = document.getElementById("totalRegistros");

  totalEl.textContent = lista.length;

  if (lista.length === 0) {
    tbody.innerHTML = `
      <tr>
        <td colspan="19" class="text-center text-muted py-4">
          Nenhum fornecedor cadastrado.
          <a href="cadFornecedores.html" class="ms-2">Cadastrar agora →</a>
        </td>
      </tr>`;
    return;
  }

  tbody.innerHTML = lista.map((c) => `
    <tr>
      <td>${escaparHTML(c.id)}</td>
      <td class="text-nowrap">
        <button class="btn btn-sm btn-success py-0 px-1 me-1" title="Editar"
          onclick="editarFornecedor(${Number(c.id)})">✏️ Editar</button>
        <button class="btn btn-sm btn-danger py-0 px-1" title="Excluir"
          onclick="excluirFornecedor(${Number(c.id)})">🗑️ Deletar</button>
      </td>
      <td style="min-width:250px">${escaparHTML(c.razaoSocial)}</td>
      <td style="min-width:180px">${escaparHTML(c.cnpj)}</td>
      <td>${escaparHTML(c.nomeFantasia)}</td>
      <td>${escaparHTML(c.ie)}</td>
      <td>${escaparHTML(c.nomeResponsavel)}</td>
      <td style="min-width:120px">${escaparHTML(c.telefone)}</td>
      <td style="min-width:140px">${escaparHTML(c.celular)}</td>
      <td>${escaparHTML(c.email)}</td>
      <td style="min-width:110px">${escaparHTML(c.cep)}</td>
      <td style="min-width:200px">${escaparHTML(c.rua)}</td>
      <td>${escaparHTML(c.numero)}</td>
      <td>${escaparHTML(c.bairro)}</td>
      <td>${escaparHTML(c.cidade)}</td>
      <td>${escaparHTML(c.estado)}</td>
      <td class="text-truncate" style="max-width:150px"
          title="${escaparHTML(c.observacoes)}">${escaparHTML(c.observacoes)}</td>
      <td class="text-nowrap">${formatarData(c.dataCadastro)}</td>
      <td class="text-nowrap">${formatarData(c.dataAlteracao)}</td>
    </tr>`).join("");
}

// ── Carregar / pesquisar ──────────────────────────────────

function carregarDadosFornecedores() {
  const termo = (document.getElementById("pesquisa")?.value || "").toLowerCase().trim();
  let lista = getFornecedores();
  if (termo) {
    lista = lista.filter((c) =>
      (c.razaoSocial && c.razaoSocial.toLowerCase().includes(termo)) ||
      (c.cnpj   && c.cnpj.replace(/\D/g, "").includes(termo.replace(/\D/g, ""))) ||
      (c.email && c.email.toLowerCase().includes(termo)) ||
      (c.cidade && c.cidade.toLowerCase().includes(termo))
    );
  }
  renderizarTabela(lista);
}

// ── Editar ────────────────────────────────────────────────

function editarFornecedor(id) {
  const lista      = getFornecedores();
  const fornecedor = lista.find((c) => Number(c.id) === Number(id));
  if (!fornecedor) { alert("Fornecedor não encontrado."); return; }
  localStorage.setItem(EDIT_KEY, JSON.stringify(fornecedor));
  window.location.href = "cadFornecedores.html?edit=" + encodeURIComponent(fornecedor.id);
}

// Variante usada pelo botão "Editar" dentro do modal de pesquisa por CNPJ:
// fecha o modal primeiro (evitando qualquer erro na transição do Bootstrap
// impedir a navegação) e só então carrega o fornecedor e redireciona.
function editarFornecedorModal(id) {
  try {
    const modalEl = document.getElementById("modalCPFNav");
    const modalInstance = modalEl && bootstrap.Modal.getInstance(modalEl);
    if (modalInstance) modalInstance.hide();
  } catch (err) {
    console.warn("[tabFornecedores] falha ao fechar modal:", err);
  }
  editarFornecedor(id);
}

// ── Excluir ───────────────────────────────────────────────

function excluirFornecedor(id) {
  const lista      = getFornecedores();
  const fornecedor = lista.find((c) => Number(c.id) === Number(id));
  const razao      = fornecedor?.razaoSocial || "—";
  const cnpj       = fornecedor?.cnpj        || "—";
  if (!confirm(`Excluir o fornecedor?\n\nRazão Social: ${razao}\nCNPJ: ${cnpj}\n\nEsta ação não pode ser desfeita.`)) return;
  saveFornecedores(lista.filter((c) => Number(c.id) !== Number(id)));
  carregarDadosFornecedores();
}

// ── Novo fornecedor ───────────────────────────────────────

function novoFornecedor() {
  localStorage.removeItem(EDIT_KEY);
  window.location.href = "cadFornecedores.html";
}

// ── Exportar CSV ──────────────────────────────────────────

function exportarCSVFornecedores() {
  const lista = getFornecedores();
  if (lista.length === 0) { alert("Nenhum fornecedor para exportar."); return; }
  const cab  = ["ID","Razão Social","CNPJ","Nome Fantasia","IE","Responsável","Telefone","Celular",
                "Email","CEP","Rua","Número","Bairro","Cidade","Estado",
                "Observações","Cadastro","Alteração"];
  const rows = lista.map((c) =>
    [c.id, c.razaoSocial, c.cnpj, c.nomeFantasia, c.ie, c.nomeResponsavel,
     c.telefone, c.celular, c.email, c.cep, c.rua, c.numero,
     c.bairro, c.cidade, c.estado, c.observacoes, c.dataCadastro, c.dataAlteracao]
    .map((v) => `"${String(v??'').replace(/"/g,'""')}"`)
    .join(";")
  );
  const csv  = [cab.map(h=>`"${h}"`).join(";"), ...rows].join("\r\n");
  const blob = new Blob(["\uFEFF"+csv], {type:"text/csv;charset=utf-8;"});
  const url  = URL.createObjectURL(blob);
  const a    = Object.assign(document.createElement("a"), {href:url, download:`fornecedores_${new Date().toISOString().slice(0,10)}.csv`});
  a.click(); URL.revokeObjectURL(url);
}

// ── Baixar JSON ───────────────────────────────────────────

function baixarJSONFornecedores() {
  const lista = getFornecedores();
  if (lista.length === 0) { alert("Nenhum fornecedor para baixar."); return; }
  const blob = new Blob([JSON.stringify(lista,null,2)], {type:"application/json"});
  const url  = URL.createObjectURL(blob);
  const a    = Object.assign(document.createElement("a"), {href:url, download:`fornecedores_${new Date().toISOString().slice(0,10)}.json`});
  a.click(); URL.revokeObjectURL(url);
}

// ── Importar JSON ─────────────────────────────────────────

function importarJSONFornecedores(event) {
  const file = event.target.files[0];
  if (!file) return;
  const reader = new FileReader();
  reader.onload = function(e) {
    try {
      const dados = JSON.parse(e.target.result);
      if (!Array.isArray(dados)) throw new Error("O JSON não é uma lista de fornecedores.");
      const existentes = getFornecedores();
      const ids = new Set(existentes.map(c => Number(c.id)));
      let add = 0, upd = 0;
      dados.forEach(novo => {
        if (!novo.id) return;
        if (ids.has(Number(novo.id))) {
          const idx = existentes.findIndex(c => Number(c.id) === Number(novo.id));
          existentes[idx] = {...existentes[idx], ...novo};
          upd++;
        } else {
          existentes.push(novo); add++;
        }
      });
      saveFornecedores(existentes);
      carregarDadosFornecedores();
      alert(`Importação concluída!\n✅ Adicionados: ${add}\n🔄 Atualizados: ${upd}`);
    } catch(err) {
      alert("Erro ao importar JSON:\n" + err.message);
    } finally {
      event.target.value = "";
    }
  };
  reader.readAsText(file, "UTF-8");
}


// ── Pesquisa CNPJ Navbar ───────────────────────────────────

function mascaraCNPJNav(input) {
  // Formato CNPJ: 00.000.000/0000-00
  let v = input.value.replace(/\D/g, "").slice(0, 14);
  if (v.length > 12)      v = v.replace(/(\d{2})(\d{3})(\d{3})(\d{4})(\d{1,2})/, "$1.$2.$3/$4-$5");
  else if (v.length > 8)  v = v.replace(/(\d{2})(\d{3})(\d{3})(\d{1,4})/, "$1.$2.$3/$4");
  else if (v.length > 5)  v = v.replace(/(\d{2})(\d{3})(\d{1,3})/, "$1.$2.$3");
  else if (v.length > 2)  v = v.replace(/(\d{2})(\d{1,3})/, "$1.$2");
  input.value = v;
}
